1//! Multi-region schema check for SeaORM entities — the driver half.
2//!
3//! [`describe`] turns a SeaORM entity marked with `#[derive(CrdbLocality)]` into a
4//! [`MultiregionTable`] descriptor; [`verify_multiregion`] loads the live schema and
5//! runs the pure [`cockroachdb_core::multiregion::evaluate`] diff. This mirrors the
6//! Spring Boot starter's startup check, on the SeaORM path.
7//!
8//! It is **opt-in**: call [`verify_multiregion`] after [`crate::connect`], passing the
9//! entities to check. In [`Mode::Warn`] (the default) it logs problems and continues;
10//! in [`Mode::Validate`], or on a contradiction in any mode but [`Mode::None`], it
11//! returns `Err` so the caller can fail startup.
12//!
13//! ```no_run
14//! # use sea_orm::DatabaseConnection;
15//! # use cockroachdb_seaorm::{describe, verify_multiregion, Mode};
16//! # mod ideas { use sea_orm::entity::prelude::*;
17//! #   #[derive(Clone, Debug, PartialEq, DeriveEntityModel)]
18//! #   #[sea_orm(table_name = "ideas")] pub struct Model { #[sea_orm(primary_key)] pub id: i64 }
19//! #   #[derive(Copy, Clone, Debug, EnumIter, DeriveRelation)] pub enum Relation {}
20//! #   impl ActiveModelBehavior for ActiveModel {}
21//! #   impl ::cockroachdb_core::multiregion::CrdbLocality for Entity {
22//! #     fn crdb_locality() -> ::cockroachdb_core::multiregion::DeclaredLocality {
23//! #       ::cockroachdb_core::multiregion::DeclaredLocality::Global } } }
24//! # async fn demo(db: DatabaseConnection) -> Result<(), sea_orm::DbErr> {
25//! let report = verify_multiregion(&db, Mode::Warn, vec![describe::<ideas::Entity>()]).await?;
26//! assert!(report.is_empty() || !report.has_contradictions());
27//! # Ok(()) }
28//! ```

44/// Describes a marked entity for the check: its table, schema, declared locality, and
45/// region column. `E` must be a SeaORM entity with `#[derive(CrdbLocality)]`.
46pub fn describe<E>() -> MultiregionTable
47where
48    E: EntityTrait + CrdbLocality,
49{
50    let entity = E::default();
51    let (kind, region_column) = match E::crdb_locality() {
52        DeclaredLocality::Global => (Kind::Global, None),
53        DeclaredLocality::RegionalByRow { region_column } => (Kind::RegionalByRow, region_column),
54    };
55    MultiregionTable {
56        entity: std::any::type_name::<E>().to_string(),
57        schema: entity.schema_name().map(str::to_string),
58        table: entity.table_name().to_string(),
59        kind,
60        region_column,
61    }
62}

64/// Describes an unmarked entity, so the check can also report a `GLOBAL` or
65/// `REGIONAL BY ROW` table whose entity is missing its marker (the reverse
66/// direction). `E` only needs to be a SeaORM entity.
67pub fn describe_plain<E>() -> MultiregionTable
68where
69    E: EntityTrait,
70{
71    let entity = E::default();
72    MultiregionTable {
73        entity: std::any::type_name::<E>().to_string(),
74        schema: entity.schema_name().map(str::to_string),
75        table: entity.table_name().to_string(),
76        kind: Kind::None,
77        region_column: None,
78    }
79}

81/// Checks `tables` against the live multi-region schema of `db`.
82///
83/// Returns an empty report when `mode` is [`Mode::None`], when `tables` is empty, or
84/// when the database is not CockroachDB (the check skips quietly). Otherwise loads the
85/// schema, diffs it, and — if there are problems — either returns `Err` (validate, or
86/// any contradiction) or logs to stderr and returns the report (warn).
87pub async fn verify_multiregion(
88    db: &DatabaseConnection,
89    mode: Mode,
90    tables: Vec<MultiregionTable>,
91) -> Result<MultiregionReport, DbErr> {
92    if mode == Mode::None || tables.is_empty() {
93        return Ok(MultiregionReport::default());
94    }
95    if !is_cockroachdb(db).await? {
96        eprintln!("CockroachDB multiregion check skipped: the database is not CockroachDB");
97        return Ok(MultiregionReport::default());
98    }
99    let state = load(db, &tables).await?;
100    let report = evaluate(&state, &tables);
101    if report.is_empty() {
102        return Ok(report);
103    }
104    if fails_startup(mode, &report) {
105        return Err(DbErr::Custom(report.format(mode)));
106    }
107    eprintln!("{}", report.format(mode));
108    Ok(report)
109}

111/// Whether `db` is CockroachDB. The check skips other databases quietly.
112async fn is_cockroachdb(db: &DatabaseConnection) -> Result<bool, DbErr> {
113    let row = db
114        .query_one(Statement::from_string(PG, "SELECT version()"))
115        .await?;
116    Ok(row
117        .and_then(|r| r.try_get_by::<String, _>(0usize).ok())
118        .map(|v| v.contains("CockroachDB"))
119        .unwrap_or(false))
120}

122fn name_of(table: &MultiregionTable, current_schema: &str) -> TableName {
123    TableName {
124        schema: table
125            .schema
126            .clone()
127            .unwrap_or_else(|| current_schema.to_string()),
128        table: table.table.clone(),
129    }
130}

132/// Reads what [`evaluate`] needs. Uses `SHOW` statements and `information_schema`,
133/// never `crdb_internal`, matching the Spring starter.
134async fn load(
135    db: &DatabaseConnection,
136    tables: &[MultiregionTable],
137) -> Result<MultiregionState, DbErr> {
138    let row = db
139        .query_one(Statement::from_string(
140            PG,
141            "SELECT current_database(), current_schema()",
142        ))
143        .await?
144        .ok_or_else(|| {
145            DbErr::Custom("multiregion check: current_database() returned no row".into())
146        })?;
147    let database: String = row.try_get_by::<String, _>(0usize)?;
148    let current_schema: String = row
149        .try_get_by::<Option<String>, _>(1usize)?
150        .unwrap_or_else(|| "public".to_string());
151
152    let primary_region = db
153        .query_one(Statement::from_sql_and_values(
154            PG,
155            "SELECT primary_region FROM [SHOW DATABASES] WHERE database_name = $1",
156            [Value::from(database.clone())],
157        ))
158        .await?
159        .and_then(|r| r.try_get_by::<Option<String>, _>(0usize).ok().flatten());
160
161    let mut cluster_regions = Vec::new();
162    for r in db
163        .query_all(Statement::from_string(
164            PG,
165            "SELECT region FROM [SHOW REGIONS FROM CLUSTER] ORDER BY region",
166        ))
167        .await?
168    {
169        cluster_regions.push(r.try_get_by::<String, _>(0usize)?);
170    }
171
172    let mut keywords = HashSet::new();
173    for r in db
174        .query_all(Statement::from_string(
175            PG,
176            "SELECT word FROM pg_get_keywords() WHERE catcode IN ('R', 'T')",
177        ))
178        .await?
179    {
180        keywords.insert(r.try_get_by::<String, _>(0usize)?);
181    }
182
183    let wanted: HashSet<TableName> = tables.iter().map(|t| name_of(t, &current_schema)).collect();
184
185    let mut found: HashMap<TableName, TableInfo> = HashMap::new();
186    for r in db
187        .query_all(Statement::from_string(
188            PG,
189            "SELECT schema_name, table_name, locality FROM [SHOW TABLES] WHERE type = 'table'",
190        ))
191        .await?
192    {
193        let name = TableName {
194            schema: r.try_get_by::<String, _>(0usize)?,
195            table: r.try_get_by::<String, _>(1usize)?,
196        };
197        if wanted.contains(&name) {
198            found.insert(
199                name,
200                TableInfo {
201                    locality: r.try_get_by::<Option<String>, _>(2usize)?,
202                    column_types: HashMap::new(),
203                },
204            );
205        }
206    }
207
208    for table in tables {
209        if table.kind != Kind::RegionalByRow {
210            continue;
211        }
212        let name = name_of(table, &current_schema);
213        if !found.contains_key(&name) {
214            continue;
215        }
216        let mut types = HashMap::new();
217        for r in db
218            .query_all(Statement::from_sql_and_values(
219                PG,
220                "SELECT column_name, crdb_sql_type FROM information_schema.columns \
221                 WHERE table_catalog = current_database() AND table_schema = $1 AND table_name = $2",
222                [
223                    Value::from(name.schema.clone()),
224                    Value::from(name.table.clone()),
225                ],
226            ))
227            .await?
228        {
229            types.insert(
230                r.try_get_by::<String, _>(0usize)?,
231                r.try_get_by::<String, _>(1usize)?,
232            );
233        }
234        if let Some(info) = found.get_mut(&name) {
235            info.column_types = types;
236        }
237    }
238
239    Ok(MultiregionState {
240        database,
241        current_schema,
242        primary_region,
243        cluster_regions,
244        tables: found,
245        keywords,
246    })
