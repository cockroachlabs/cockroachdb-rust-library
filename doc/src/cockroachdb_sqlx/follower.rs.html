1//! Follower reads via transaction-scoped `AS OF SYSTEM TIME`.
2//!
3//! The clause scopes the whole transaction (`SET TRANSACTION AS OF SYSTEM TIME
4//! …`) rather than being appended to a query string — the latter is a syntax
5//! error in CockroachDB and keeps bound parameters working.
6
7use crate::pool::Pool;
8use cockroachdb_core::follower_read_clause;
9use sqlx::postgres::PgRow;
10use std::future::Future;
11use std::pin::Pin;
12use std::time::{Duration, Instant};
13
14type FollowerBody<'c, T> = Pin<Box<dyn Future<Output = Result<T, sqlx::Error>> + Send + 'c>>;
15
16impl Pool {
17    /// Runs `f` inside a transaction scoped to a follower-read timestamp.
18    ///
19    /// `staleness == Duration::ZERO` uses `follower_read_timestamp()`; a positive
20    /// value uses a fixed `-<staleness>` interval. Bound parameters work normally.
21    pub async fn follower_read_tx<T, F>(&self, staleness: Duration, f: F) -> Result<T, sqlx::Error>
22    where
23        F: for<'c> Fn(&'c mut sqlx::PgConnection) -> FollowerBody<'c, T> + Send + Sync,
24    {
25        let clause =
26            follower_read_clause(staleness).map_err(|e| sqlx::Error::Protocol(e.to_string()))?;
27        let start = Instant::now();
28        let mut tx = self.pool.begin().await?;
29
30        let set = format!("SET TRANSACTION {}", clause);
31        if let Err(e) = sqlx::query(&set).execute(&mut *tx).await {
32            let _ = tx.rollback().await;
33            return Err(e);
34        }
35
36        match f(&mut tx).await {
37            Ok(value) => {
38                tx.commit().await?;
39                if let Some(m) = &self.metrics {
40                    m.record_follower_read(start.elapsed());
41                }
42                Ok(value)
43            }
44            Err(e) => {
45                let _ = tx.rollback().await;
46                Err(e)
47            }
48        }
49    }
50
51    /// Convenience: runs a parameterless `query` as a follower read using the
52    /// pool's default staleness and returns all rows.
53    pub async fn fetch_all_follower_read(&self, query: &str) -> Result<Vec<PgRow>, sqlx::Error> {
54        self.fetch_all_follower_read_as(self.config.staleness, query)
55            .await
56    }
57
58    /// Like [`Pool::fetch_all_follower_read`] but with an explicit staleness.
59    pub async fn fetch_all_follower_read_as(
60        &self,
61        staleness: Duration,
62        query: &str,
63    ) -> Result<Vec<PgRow>, sqlx::Error> {
64        let q = query.to_string();
65        self.follower_read_tx(staleness, move |conn| {
66            let q = q.clone();
67            Box::pin(async move { sqlx::query(&q).fetch_all(&mut *conn).await })
68        })
69        .await
70    }
