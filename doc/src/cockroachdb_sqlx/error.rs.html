1//! SQLSTATE classification for sqlx errors.
2//!
3//! The orphan rule prevents implementing [`cockroachdb_core::SqlState`] for the
4//! foreign `sqlx::Error`, so classification is exposed as free functions —
5//! matching the Go library's `pgxadapter.IsRetryable`.
6
7/// Extracts the Postgres `SQLSTATE` code from a sqlx error, if it carries one.
8///
9/// Only `sqlx::Error::Database` errors have a code; everything else returns
10/// `None`.
11pub fn sql_state(err: &sqlx::Error) -> Option<String> {
12    match err {
13        sqlx::Error::Database(db) => db.code().map(|c| c.into_owned()),
14        _ => None,
15    }
16}
17
18/// Returns `true` if the error is a retryable CockroachDB conflict
19/// (`40001` serialization failure or `40P01` deadlock).
20pub fn is_retryable(err: &sqlx::Error) -> bool {
21    sql_state(err)
22        .as_deref()
23        .map(cockroachdb_core::is_retryable_code)
24        .unwrap_or(false)
25}
