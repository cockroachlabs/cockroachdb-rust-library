1//! SQLx adapter for CockroachDB — the primary adapter for this library.
2//!
3//! Wraps a [`sqlx::PgPool`] with CockroachDB-aware behaviour:
4//!
5//! * [`Pool::execute_tx`] runs a closure in a transaction and retries it on
6//!   serialization conflicts (`SQLSTATE 40001`) with exponential backoff.
7//! * [`Pool::follower_read_tx`] / [`Pool::fetch_all_follower_read`] run reads
8//!   against a transaction scoped with `AS OF SYSTEM TIME`.
9//! * Isolation level and read-only mode are applied per transaction.
10//! * The pool implements [`cockroachdb_core::HealthChecker`].
11//!
12//! Re-runnable closures take `&mut sqlx::PgConnection` and return a boxed future;
13//! this works around Rust's lending-closure limits. A typical call:
14//!
15//! ```no_run
16//! # use cockroachdb_sqlx::Pool;
17//! # use cockroachdb_core::Config;
18//! # async fn demo(pool: Pool) -> Result<(), sqlx::Error> {
19//! pool.execute_tx(|conn| Box::pin(async move {
20//!     sqlx::query("UPDATE accounts SET balance = balance - 100 WHERE id = $1")
21//!         .bind(1_i64)
22//!         .execute(&mut *conn)
23//!         .await?;
24//!     Ok(())
25//! })).await
26//! # }
27//! ```
