1//! Metrics for the webhook sink.
2
3use std::sync::atomic::{AtomicU64, Ordering};
4use std::time::Duration;
5
6/// A point-in-time view of [`CdcMetrics`].
7#[derive(Debug, Default, Clone, PartialEq, Eq)]
8pub struct CdcSnapshot {
9    /// Row-change events dispatched to the handler.
10    pub events: u64,
11    /// Resolved-timestamp heartbeats received.
12    pub resolved: u64,
13    /// Handler invocations that returned an error.
14    pub errors: u64,
15    /// Requests rejected for failing authentication.
16    pub auth_failures: u64,
17    /// Summed handler latency.
18    pub total_latency: Duration,
19}
20
21/// Lock-free counters for webhook deliveries.
22#[derive(Debug, Default)]
23pub struct CdcMetrics {
24    events: AtomicU64,
25    resolved: AtomicU64,
26    errors: AtomicU64,
27    auth_failures: AtomicU64,
28    total_latency_nanos: AtomicU64,
29}
30
31impl CdcMetrics {
32    /// Creates an empty metrics recorder.
33    pub fn new() -> Self {
34        Self::default()
35    }
36
37    pub(crate) fn record_event(&self, latency: Duration, is_error: bool) {
38        self.events.fetch_add(1, Ordering::Relaxed);
39        if is_error {
40            self.errors.fetch_add(1, Ordering::Relaxed);
41        }
42        self.total_latency_nanos
43            .fetch_add(latency.as_nanos() as u64, Ordering::Relaxed);
44    }
45
46    pub(crate) fn record_resolved(&self) {
47        self.resolved.fetch_add(1, Ordering::Relaxed);
48    }
49
50    pub(crate) fn record_auth_failure(&self) {
51        self.auth_failures.fetch_add(1, Ordering::Relaxed);
52    }
53
54    /// Returns a snapshot of the current counters.
55    pub fn snapshot(&self) -> CdcSnapshot {
56        CdcSnapshot {
57            events: self.events.load(Ordering::Relaxed),
58            resolved: self.resolved.load(Ordering::Relaxed),
59            errors: self.errors.load(Ordering::Relaxed),
60            auth_failures: self.auth_failures.load(Ordering::Relaxed),
61            total_latency: Duration::from_nanos(self.total_latency_nanos.load(Ordering::Relaxed)),
62        }
63    }
64}
