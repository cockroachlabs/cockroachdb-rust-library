1//! Change data capture (CDC) for CockroachDB.
2//!
3//! Three driver-agnostic pieces mirroring the Go library's `cdc` package:
4//!
5//! * [`ChangefeedSpecBuilder`] builds a `CREATE CHANGEFEED` statement — run the
6//!   resulting SQL with your own pool (this crate stays decoupled from any driver).
7//! * [`Envelope`] / [`Payload`] parse the JSON a CockroachDB webhook sink posts.
8//! * [`WebhookSink`] validates optional bearer auth, parses the envelope, and
9//!   dispatches each row to your [`EventHandler`], recording [`CdcMetrics`].
10
11mod changefeed;
12mod event;
13mod handler;
14mod metrics;
15
16pub use changefeed::{ChangefeedSpec, ChangefeedSpecBuilder};
17pub use event::{Envelope, Operation, Payload};
18pub use handler::{EventHandler, Outcome, WebhookSink};
19pub use metrics::{CdcMetrics, CdcSnapshot};

21/// A boxed error returned by an [`EventHandler`].
22pub type BoxError = Box<dyn std::error::Error + Send + Sync>;
23
24/// Errors from building changefeeds or handling webhook deliveries.
25#[derive(Debug, thiserror::Error)]
26pub enum CdcError {
27    /// A changefeed spec was missing its target table.
28    #[error("changefeed requires a table")]
29    MissingTable,
30    /// A changefeed spec was missing its sink URI.
31    #[error("changefeed requires a sink (webhook target)")]
32    MissingSink,
33    /// A table or column identifier was empty.
34    #[error("invalid identifier: {0:?}")]
35    InvalidIdentifier(String),
36    /// The webhook request failed bearer-token authentication.
37    #[error("unauthorized webhook request")]
38    Unauthorized,
39    /// The webhook body could not be parsed as a changefeed envelope.
40    #[error("invalid changefeed payload: {0}")]
41    InvalidPayload(#[from] serde_json::Error),
42    /// The registered event handler returned an error.
43    #[error("event handler failed: {0}")]
44    Handler(#[source] BoxError),
45}
