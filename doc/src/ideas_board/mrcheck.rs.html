1//! Multi-region annotation check (SeaORM path): declare each entity's intended
2//! CockroachDB locality with `#[derive(CrdbLocality)]`, then verify that intent
3//! against the live schema at startup. This is the Rust twin of the Spring Boot
4//! starter's startup multi-region check (0.2.0). It is opt-in: call [`verify`]
5//! after connecting.
6
7// tag::mr-check-entity[]
8/// The `ideas` table modelled as a SeaORM entity, declared **REGIONAL BY ROW**
9/// and homed by its `region` column. `#[derive(CrdbLocality)]` records that
10/// intent so [`verify`] can diff it against the live table's locality.
11pub mod regional {
12    use cockroachdb_macros::CrdbLocality;
13    use sea_orm::entity::prelude::*;
14
15    #[derive(Clone, Debug, PartialEq, DeriveEntityModel, CrdbLocality)]
16    #[sea_orm(table_name = "ideas")]
17    #[crdb(regional_by_row)]
18    pub struct Model {
19        #[sea_orm(primary_key)]
20        pub id: i64,
21        pub title: String,
22        #[crdb(region)]
23        pub region: String,
24    }
25
26    #[derive(Copy, Clone, Debug, EnumIter, DeriveRelation)]
27    pub enum Relation {}
28
29    impl ActiveModelBehavior for ActiveModel {}
30}
31// end::mr-check-entity[]
32
33// tag::mr-check-usage[]
34use cockroachdb_seaorm::{describe, verify_multiregion, Mode};
35use sea_orm::DatabaseConnection;
36
37/// Verifies every annotated entity's declared locality against the live schema.
38/// In `Mode::Warn` (the default) a plain mismatch — e.g. the `ideas` table not
39/// being multi-region on a single-node cluster — is logged and startup
40/// continues; a direct contradiction fails startup even in warn mode.
41pub async fn verify(db: &DatabaseConnection) -> Result<(), sea_orm::DbErr> {
42    let report = verify_multiregion(db, Mode::Warn, vec![describe::<regional::Entity>()]).await?;
43    for problem in report.problems() {
44        eprintln!("multi-region check: {problem}");
45    }
46    Ok(())
47}
48// end::mr-check-usage[]

50#[cfg(test)]
51mod tests {
52    use super::{describe, regional};
53    use cockroachdb_seaorm::Kind;
54
55    #[test]
56    fn entity_declares_regional_by_row_homed_by_region() {
57        let t = describe::<regional::Entity>();
58        assert_eq!(t.table, "ideas");
59        assert_eq!(t.kind, Kind::RegionalByRow);
60        assert_eq!(t.region_column.as_deref(), Some("region"));
61    }
62
63    // tag::mr-check-report[]
64    // End-to-end check against a live cluster. On the single-node CI cluster the
65    // `ideas` table has no regions, so `verify` reports a plain "not multi-region"
66    // problem and (in warn mode) returns Ok — nothing to fail on. `#[ignore]`d;
67    // the repo runs integration tests with `-- --ignored`.
68    use super::verify;
69    use cockroachdb_core::Config;
70    use cockroachdb_seaorm::connect;
71
72    #[tokio::test(flavor = "multi_thread")]
73    #[ignore = "requires a live CockroachDB cluster (set CRDB_URL)"]
74    async fn verify_runs_against_live_schema() {
75        let url = std::env::var("CRDB_URL")
76            .unwrap_or_else(|_| "postgres://root@localhost:26257/defaultdb?sslmode=disable".into());
77        let db = connect(&url, &Config::default()).await.expect("connect");
78        // Warn mode tolerates a non-multi-region table, so this is Ok on a
79        // single-node cluster with no regions configured.
80        verify(&db).await.expect("warn-mode verify returns Ok");
81    }
82    // end::mr-check-report[]
83}
