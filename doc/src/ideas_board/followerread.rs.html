1//! Follower-reads track: serve the leaderboard from the nearest replica by
2//! scoping the read to a slightly-stale timestamp.
3
4// tag::follower-read[]
5use crate::idea::Idea;
6use cockroachdb_sqlx::Pool;
7use sqlx::Row;
8use std::time::Duration;
9
10/// Returns the top ideas by votes, served from a **follower read**.
11/// `follower_read_tx` scopes the query with `AS OF SYSTEM TIME`, so CockroachDB
12/// can answer it from the nearest replica instead of the leaseholder — lower
13/// latency, at the cost of reading data up to `staleness` old. Pass
14/// `Duration::ZERO` to let CockroachDB pick the freshest safe timestamp.
15pub async fn top_ideas_follower(
16    pool: &Pool,
17    staleness: Duration,
18    limit: i64,
19) -> Result<Vec<Idea>, sqlx::Error> {
20    pool.follower_read_tx(staleness, move |conn| {
21        Box::pin(async move {
22            let rows = sqlx::query(
23                "SELECT id::text AS id, title, votes FROM ideas ORDER BY votes DESC LIMIT $1",
24            )
25            .bind(limit)
26            .fetch_all(&mut *conn)
27            .await?;
28            Ok(rows
29                .iter()
30                .map(|r| Idea {
31                    id: r.get("id"),
32                    title: r.get("title"),
33                    votes: r.get("votes"),
34                })
35                .collect())
36        })
37    })
38    .await
39}
40// end::follower-read[]

42#[cfg(test)]
43mod tests {
44    // tag::follower-read-it[]
45    // Follower reads are ~4.8s stale, so freshly-seeded rows are invisible until
46    // the staleness window passes. Seed, then poll until the snapshot catches up
47    // (tolerating the transient "as of a time before the table existed" error).
48    // `#[ignore]`d — needs a live cluster.
49    use super::top_ideas_follower;
50    use crate::service::IdeaService;
51    use cockroachdb_core::Config;
52    use cockroachdb_sqlx::Pool;
53    use std::time::{Duration, Instant};
54
55    #[tokio::test(flavor = "multi_thread")]
56    #[ignore = "requires a live CockroachDB cluster (set CRDB_URL)"]
57    async fn follower_read_catches_up_to_seeded_ideas() {
58        let url = std::env::var("CRDB_URL")
59            .unwrap_or_else(|_| "postgres://root@localhost:26257/defaultdb?sslmode=disable".into());
60        let pool = Pool::connect(&url, Config::default())
61            .await
62            .expect("connect");
63        let svc = IdeaService::new(pool.clone());
64        svc.ensure_schema().await.expect("schema");
65        for title in ["Dark mode", "CSV export", "Full-text search"] {
66            svc.create_idea(title).await.expect("seed");
67        }
68
69        let deadline = Instant::now() + Duration::from_secs(30);
70        let mut seen = 0;
71        while Instant::now() < deadline {
72            if let Ok(top) = top_ideas_follower(&pool, Duration::from_secs(5), 10).await {
73                seen = top.len();
74                if seen >= 3 {
75                    break;
76                }
77            }
78            tokio::time::sleep(Duration::from_millis(500)).await;
79        }
80        assert!(
81            seen >= 3,
82            "follower read should catch up to the 3 seeded ideas within 30s"
83        );
84        pool.close().await;
85    }
86    // end::follower-read-it[]
87}
