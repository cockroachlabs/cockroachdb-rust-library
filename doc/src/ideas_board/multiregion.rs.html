1//! Multi-region track: home each idea to a region via the hidden `crdb_region`
2//! **column**. CockroachDB homes a REGIONAL BY ROW row by the value written to
3//! that column on insert — there is no `SET crdb_region` session statement (a
4//! mistake that bit several sibling libraries). This module only ever writes the
5//! column value.
6
7// tag::region-config[]
8use cockroachdb_core::ddl;
9
10/// DDL that turns the board into a multi-region, REGIONAL BY ROW deployment:
11/// declare the database's regions, then set the table's locality so every row
12/// carries a `crdb_region` homing column. Run these once per cluster (adding a
13/// non-primary region needs nodes in that locality).
14pub fn regional_setup_sql(database: &str) -> Vec<String> {
15    vec![
16        ddl::set_primary_region(database, "us-east1"),
17        ddl::add_region(database, "us-west1"),
18        ddl::set_locality_regional_by_row("ideas"),
19    ]
20}
21// end::region-config[]
22
23// tag::inregion-usage[]
24use cockroachdb_core::region::{RegionContext, CRDB_REGION_COLUMN};
25use cockroachdb_sqlx::Pool;
26use sqlx::Row;
27
28/// Writes ideas homed to a caller-chosen region. The region travels as the row's
29/// `crdb_region` **column value** on insert, so CockroachDB stores and serves
30/// that row from the matching region. Homing is per-row data — there is no
31/// session-level region switch.
32#[derive(Clone)]
33pub struct RegionalIdeaService {
34    pool: Pool,
35}
36
37impl RegionalIdeaService {
38    /// Creates the service over a pool.
39    pub fn new(pool: Pool) -> Self {
40        Self { pool }
41    }
42
43    /// Inserts an idea homed to `region`, returning its id. The `(column, value)`
44    /// pair comes from `RegionContext::homing_column()` — the column name is the
45    /// fixed `crdb_region`, never caller input, so interpolating it is safe.
46    pub async fn add_in_region(&self, region: &str, title: &str) -> Result<String, sqlx::Error> {
47        let ctx = RegionContext::new(region);
48        let (col, value) = ctx.homing_column().expect("region is set");
49        let sql = format!("INSERT INTO ideas (title, {col}) VALUES ($1, $2) RETURNING id::text");
50        let (title, value) = (title.to_string(), value.to_string());
51        self.pool
52            .execute_tx(move |conn| {
53                let (sql, title, value) = (sql.clone(), title.clone(), value.clone());
54                Box::pin(async move {
55                    let row = sqlx::query(&sql)
56                        .bind(&title)
57                        .bind(&value)
58                        .fetch_one(&mut *conn)
59                        .await?;
60                    row.try_get::<String, _>(0)
61                })
62            })
63            .await
64    }
65
66    /// Reads back the region a row is homed to.
67    pub async fn region_of(&self, id: &str) -> Result<String, sqlx::Error> {
68        let sql = format!("SELECT {CRDB_REGION_COLUMN}::text FROM ideas WHERE id = $1");
69        sqlx::query(&sql)
70            .bind(id)
71            .fetch_one(self.pool.raw_pool())
72            .await?
73            .try_get::<String, _>(0)
74    }
75}
76// end::inregion-usage[]

78#[cfg(test)]
79mod tests {
80    // Offline: the setup DDL and the homing pair are pure string construction,
81    // so assert them without a cluster. The end-to-end homing proof is the
82    // `#[ignore]`d integration test below (needs a multi-region cluster).
83    use super::regional_setup_sql;
84    use cockroachdb_core::region::RegionContext;
85
86    #[test]
87    fn setup_sql_declares_regions_and_row_locality() {
88        let sql = regional_setup_sql("defaultdb").join("\n");
89        assert!(sql.contains("PRIMARY REGION"));
90        assert!(sql.contains("us-east1"));
91        assert!(sql.contains("ADD REGION"));
92        assert!(sql.contains("REGIONAL BY ROW"));
93    }
94
95    #[test]
96    fn homing_writes_the_crdb_region_column() {
97        let ctx = RegionContext::new("us-west1");
98        assert_eq!(ctx.homing_column(), Some(("crdb_region", "us-west1")));
99    }
100
101    // tag::region-test[]
102    // End-to-end homing proof. Needs a REGIONAL BY ROW cluster whose regions
103    // include us-east1/us-west1 (set CRDB_URL); `#[ignore]`d otherwise. Writes an
104    // idea in each region and reads the `crdb_region` column back to confirm the
105    // row landed where it was homed.
106    use super::RegionalIdeaService;
107    use crate::service::IdeaService;
108    use cockroachdb_core::Config;
109    use cockroachdb_sqlx::Pool;
110
111    #[tokio::test(flavor = "multi_thread")]
112    #[ignore = "requires a multi-region CockroachDB cluster (set CRDB_URL)"]
113    async fn writes_home_rows_in_bound_region() {
114        let url = std::env::var("CRDB_URL")
115            .unwrap_or_else(|_| "postgres://root@localhost:26257/defaultdb?sslmode=disable".into());
116        let pool = Pool::connect(&url, Config::default())
117            .await
118            .expect("connect");
119        IdeaService::new(pool.clone())
120            .ensure_schema()
121            .await
122            .expect("schema");
123        for stmt in regional_setup_sql("defaultdb") {
124            sqlx::query(&stmt)
125                .execute(pool.raw_pool())
126                .await
127                .expect("regional setup");
128        }
129
130        let svc = RegionalIdeaService::new(pool.clone());
131        let east = svc
132            .add_in_region("us-east1", "East idea")
133            .await
134            .expect("east");
135        let west = svc
136            .add_in_region("us-west1", "West idea")
137            .await
138            .expect("west");
139
140        assert_eq!(svc.region_of(&east).await.expect("region"), "us-east1");
141        assert_eq!(svc.region_of(&west).await.expect("region"), "us-west1");
142        pool.close().await;
143    }
144    // end::region-test[]
145}
