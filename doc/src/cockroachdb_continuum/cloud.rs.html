1//! CockroachDB Cloud API client: list clusters and export metrics.
2//!
3//! Defaults to a simulated client; networked transport is reserved for a future
4//! `http` feature. Selection is keyed on the API key, mirroring the Go library.
5
6use crate::BoxError;
7use async_trait::async_trait;
8use serde::{Deserialize, Serialize};
9
10/// A CockroachDB Cloud cluster.
11#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
12pub struct Cluster {
13    /// Cluster id.
14    pub id: String,
15    /// Cluster name.
16    pub name: String,
17    /// Cluster state (e.g. `CREATED`).
18    pub state: String,
19}
20
21/// Cloud configuration. No API key ⇒ simulated client.
22#[derive(Debug, Clone, Default)]
23pub struct CloudConfig {
24    /// API key. When `None`, the simulated client is used.
25    pub api_key: Option<String>,
26    /// Optional API endpoint override.
27    pub endpoint: Option<String>,
28}
29
30impl CloudConfig {
31    fn use_simulated(&self) -> bool {
32        self.api_key.is_none()
33    }
34}
35
36/// The Cloud client surface.
37#[async_trait]
38pub trait Cloud: Send + Sync {
39    /// Lists clusters.
40    async fn list_clusters(&self) -> Result<Vec<Cluster>, BoxError>;
41    /// Fetches one cluster by id.
42    async fn get_cluster(&self, id: &str) -> Result<Cluster, BoxError>;
43    /// Exports a single metric sample to Cloud.
44    async fn export_metric(&self, name: &str, value: f64) -> Result<(), BoxError>;
45}
46
47/// Offline Cloud client with one canned cluster; records exported metrics.
48#[derive(Debug, Default)]
49pub struct Simulated {
50    exported: std::sync::Mutex<Vec<(String, f64)>>,
51}
52
53impl Simulated {
54    /// Metrics exported so far (for tests/inspection).
55    pub fn exported(&self) -> Vec<(String, f64)> {
56        self.exported.lock().unwrap().clone()
57    }
58
59    fn sample() -> Cluster {
60        Cluster {
61            id: "sim-cluster-1".into(),
62            name: "simulated".into(),
63            state: "CREATED".into(),
64        }
65    }
66}
67
68#[async_trait]
69impl Cloud for Simulated {
70    async fn list_clusters(&self) -> Result<Vec<Cluster>, BoxError> {
71        Ok(vec![Self::sample()])
72    }
73
74    async fn get_cluster(&self, id: &str) -> Result<Cluster, BoxError> {
75        let mut c = Self::sample();
76        c.id = id.to_string();
77        Ok(c)
78    }
79
80    async fn export_metric(&self, name: &str, value: f64) -> Result<(), BoxError> {
81        self.exported
82            .lock()
83            .unwrap()
84            .push((name.to_string(), value));
85        Ok(())
86    }
87}
88
89struct NotEnabled;
90
91#[async_trait]
92impl Cloud for NotEnabled {
93    async fn list_clusters(&self) -> Result<Vec<Cluster>, BoxError> {
94        Err(http_disabled())
95    }
96    async fn get_cluster(&self, _id: &str) -> Result<Cluster, BoxError> {
97        Err(http_disabled())
98    }
99    async fn export_metric(&self, _name: &str, _value: f64) -> Result<(), BoxError> {
100        Err(http_disabled())
101    }
102}
103
104fn http_disabled() -> BoxError {
105    "continuum HTTP transport is not enabled in this build".into()
106}
107
108/// Builds a Cloud client: simulated when no API key is configured.
109pub fn new(config: CloudConfig) -> Box<dyn Cloud> {
110    if config.use_simulated() {
111        Box::new(Simulated::default())
112    } else {
113        Box::new(NotEnabled)
114    }
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120
121    #[tokio::test]
122    async fn simulated_lists_and_exports() {
123        let client = new(CloudConfig::default());
124        assert_eq!(client.list_clusters().await.unwrap().len(), 1);
125        assert_eq!(client.get_cluster("abc").await.unwrap().id, "abc");
126        client.export_metric("tx_count", 42.0).await.unwrap();
127    }
128
129    #[tokio::test]
130    async fn api_key_without_http_errors() {
131        let client = new(CloudConfig {
132            api_key: Some("k".into()),
133            endpoint: None,
134        });
135        assert!(client.list_clusters().await.is_err());
136    }
